What the edge pipeline does

Schema validation
Malformed events are caught before they leave the device.
Metadata enrichment
Every event gets device, site, model version and session context.
Class mapping
"suitcase" becomes "box" in pipeline YAML, not in application code.
Lineage tracking
Which node, which pipeline, and when the event was ingested.
Offline buffering
An NVMe-backed buffer survives network outages with no data loss.
Fleet deployment
Expanso Cloud deploys to one edge node or a thousand with one command.

Pipeline 1: YOLO GPU detection

yolo-detector-job.yaml 
name: yolo-detector
type: pipeline
config:
  input:
    subprocess:
      name: /usr/bin/docker
      args: [run, --rm, --runtime=nvidia, --network=host,
        -v, /opt/box-monitor:/data,
        -v, /opt/box-monitor/yolov8s-worldv2.pt:/root/yolov8s-worldv2.pt:ro,
        -w, /root, ultralytics/ultralytics:latest-jetson-jetpack6,
        bash, -c, "pip install -q git+https://github.com/ultralytics/CLIP.git && python3 /data/scripts/detect_loop.py"]
      codec: lines
      restart_on_exit: true
  pipeline:
    processors:
      - mapping: root = this.parse_json().catch(deleted()) # parse detection events
      - mapping: | # enrich with device context
          root = this
          root.source = { "device_id": env("DEVICE_ID"), "site_id": env("SITE_ID"), "pipeline": "yolo-detector" }
          root.detected_at = now()
  output:
    broker: { pattern: fan_out, outputs: [
      { file: { path: detection-events.ndjson, codec: lines } },
      { stdout: { codec: lines } }] }

Pipeline 2: event enrichment and routing

security-camera-events-job.yaml 
name: security-camera-events
type: pipeline
config:
  input:
    subprocess:
      name: uvx
      args: [--quiet, --from, "expanso-security-camera @ git+...@v0.1.0", esc-infer, config.yaml]
      codec: lines
      restart_on_exit: true
  pipeline:
    processors:
      - mapping: root = this.parse_json().catch(deleted()) # drop non-JSON
      - mapping: root = if this.schema_version == null { deleted() } else { this } # validate schema
      - mapping: | # enrich with device and site metadata
          root = this
          root.source = { "device_id": env("DEVICE_ID"), "site_id": env("SITE_ID"), "camera_id": this.camera_id }
          root.enriched_at = now()
      - mapping: | # map COCO classes to business labels
          let mapped = match this.detection.object_class_raw.or("") {
            "suitcase" => "box", "cardboard box" => "box", "person" => "person", _ => "object" }
          root.detection.object_class = $mapped
      - mapping: | # lineage tracking
          root.lineage = { "edge_node": env("HOSTNAME"), "pipeline": "security-camera-events", "ingested_at": now() }
  output:
    broker: { pattern: fan_out, outputs: [
      { file: { path: enriched-events.ndjson, codec: lines } },
      { stdout: { codec: lines } }] }
